#include <iostream>
using namespace std;
class Test{
private:
    int x;
public:
    Test(int a){
        x=a;
    }
    void show(){
        cout<<"Value: "<<x<<endl;
    }
};
int main(){
    Test t(42);
    t.show();
    int arr[5]={1,2,3,4,5};
    for(int i=0;i<5;i++){
        cout<<arr[i]<<" ";
    }
    cout<<endl;
    return 0;
}
