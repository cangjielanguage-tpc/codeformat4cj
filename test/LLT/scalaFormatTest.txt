import scala.util.Random
object UnformattedCode{
    def main(args:Array[String]):Unit={
        val r=new Random();
        val numbers=List.fill(10)(r.nextInt(100));
        println("Original:"+numbers);
        val filtered=numbers.filter(_>50).map(_*2).sorted;
        println("Processed:"+filtered);
        def sumList(lst:List[Int]):Int=lst match{
            case Nil=>0;
            case h::t=>h+sumList(t)
        };
        println("Sum:"+sumList(filtered))
    }
}
